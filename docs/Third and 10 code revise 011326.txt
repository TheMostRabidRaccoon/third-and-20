# play_state.py
from __future__ import annotations
from dataclasses import dataclass
from typing import Optional, Literal


DriveDir = Literal[1, -1]


@dataclass(frozen=True)
class PlayState:
    """
    Canonical, team-neutral per-play state.


    IMPORTANT:
    - Contains ONLY derived play-level state.
    - Does NOT contain raw positions, clusters, or any per-frame arrays.
    - All spatial values are in FIELD COORDINATES (yards), not normalized 0..1.
    """
    play_id: str
    snap_frame: int


    # Field-state
    los_x: float                      # line of scrimmage in yards along field length axis
    possession_team_id: str           # team with the ball for this play
    drive_dir: DriveDir               # +1 or -1 along the field x-axis


    # Confidence (0..1) for the play-state inference as a whole
    confidence: float = 1.0


    # Optional metadata for debugging / audits (keep minimal)
    source: Optional[str] = None      # e.g. "scorebug", "formation", "mixed"

from dataclasses import dataclass

from play_state import PlayState, DriveDir


@dataclass
class ResolverConfig:
    # Tweakable thresholds; keep these simple and explicit.
    min_players: int = 18
    drive_dir_min_dx_yards: float = 1.0    # minimum Δx in early post-snap window to trust drive_dir
    drive_dir_window_frames: int = 10      # frames after snap to estimate direction


class PlayStateResolver:
    """
    Resolves PlayState from tracking + (optional) OCR/scorebug signals.


    This module MUST NOT:
    - label offense/defense on tracks
    - compute metrics like box/front/shell
    - recompute LOS later using offense/defense lists
    """
    def __init__(self, cfg: Optional[ResolverConfig] = None):
        self.cfg = cfg or ResolverConfig()


    def resolve(
        self,
        play_id: str,
        snap_frame: int,
        tracking,                              # your tracking object
        teams: Tuple[str, str],                # (team_id_a, team_id_b)
        scorebug: Optional[Dict[str, Any]] = None
    ) -> PlayState:
        """
        Returns a PlayState computed ONCE per play.
        Assumes tracking positions are already in FIELD COORDS (yards).
        """


        # 1) LOS from formation geometry (team-neutral)
        los_x, los_conf = self._estimate_los_x(tracking, snap_frame)


        # 2) Possession inference (prefer scorebug, fallback to formation/ball heuristics)
        poss_id, poss_conf, source = self._infer_possession_team(tracking, snap_frame, teams, scorebug)


        # 3) Drive direction from early post-snap ball/centroid movement
        drive_dir, dir_conf = self._infer_drive_dir(tracking, snap_frame, los_x)


        # Combine confidence conservatively (min is fine; product is stricter)
        conf = min(los_conf, poss_conf, dir_conf)


        return PlayState(
            play_id=play_id,
            snap_frame=snap_frame,
            los_x=los_x,
            possession_team_id=poss_id,
            drive_dir=drive_dir,
            confidence=conf,

# metrics_presnap.py
from __future__ import annotations
from dataclasses import dataclass
from typing import Optional


from play_state import PlayState


@dataclass(frozen=True)
class Metric:
    play_id: str
    team_id: str
    name: str
    value: float
    confidence: float


def compute_box_count(tracking, play_state: PlayState, offense_team_id: str, defense_team_id: str,
                      box_depth_yards: float = 5.0, box_width_yards: float = 8.0) -> Metric:
    """
    Example: count defenders near LOS at snap.
    Geometry only. No CV. No clustering libs.
    """
    snap = play_state.snap_frame
    los_x = play_state.los_x


    defenders = []
    for pid, p in tracking.players.items():
        if getattr(p, "team_id", None) != defense_team_id:
            continue
        pt = p.get_position_at_frame(snap)
        if pt is None:
            continue
        # In field coords: "near LOS" means |x - los_x| <= box_depth
        if abs(pt.x - los_x) <= box_depth_yards:
            defenders.append(pt)


    # confidence inherits from play_state, can be refined with tracking quality later
    return Metric(
        play_id=play_state.play_id,
        team_id=defense_team_id,
        name="box_count",
        value=float(len(defenders)),
        confidence=play_state.confidence
    )

# test_invariance.py
def team_swap_invariance(metric_fn, tracking_a, tracking_b, play_state_a, play_state_b):
    """
    Swap team labels + possession accordingly; canonical metric value should match.
    """
    m1 = metric_fn(tracking_a, play_state_a)
    m2 = metric_fn(tracking_b, play_state_b)
    assert m1.value == m2.value, (m1, m2)
